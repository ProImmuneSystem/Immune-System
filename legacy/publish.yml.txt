name: Publish site and scheduled articles
on:
  push:
    branches: [main]
  schedule:
    - cron: '17 1 * * *' # 매일 한국 시간 오전 10시 17분
  workflow_dispatch:
permissions:
  contents: write
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: false
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Release one queued article on schedule
        if: github.event_name == 'schedule'
        run: |
          python scripts/release.py
          if ! git diff --quiet -- content/state.json; then
            git config user.name 'github-actions[bot]'
            git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
            git add content/state.json
            git commit -m 'Publish next scheduled article'
            git push
          fi
      - run: python scripts/build.py
      - uses: actions/configure-pages@v5
        with:
          enablement: true
      - uses: actions/upload-pages-artifact@v3
        with:
          path: site
  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
